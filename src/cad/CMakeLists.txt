add_library(medforge_cad STATIC src/Module.cpp src/Kernel.cpp)
add_library(medforge::cad ALIAS medforge_cad)
target_include_directories(medforge_cad PUBLIC include)
target_include_directories(medforge_cad PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(medforge_cad PUBLIC medforge::geometry)
target_compile_features(medforge_cad PUBLIC cxx_std_20)
set_target_properties(medforge_cad PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
medforge_set_core_warnings(medforge_cad)

set(_medforge_occt_resource "")
set(_medforge_occt_candidates)
if(WIN32 AND MSVC)
    list(APPEND _medforge_occt_candidates
        "${CMAKE_SOURCE_DIR}/third_party/dist/occt-7.9.3-windows-msvc-debug/cmake")
else()
    list(APPEND _medforge_occt_candidates
        "${CMAKE_SOURCE_DIR}/third_party/dist/occt-7.9.3-linux/lib/cmake/opencascade")
endif()
if(NOT OpenCASCADE_DIR)
    foreach(_medforge_occt_candidate IN LISTS _medforge_occt_candidates)
        if(EXISTS "${_medforge_occt_candidate}/OpenCASCADEConfig.cmake")
            set(OpenCASCADE_DIR "${_medforge_occt_candidate}" CACHE PATH "Open CASCADE cmake package" FORCE)
            break()
        endif()
    endforeach()
endif()
find_package(OpenCASCADE 7.9.3 CONFIG QUIET)
if(OpenCASCADE_FOUND)
    message(STATUS "Open CASCADE ${OpenCASCADE_MAJOR_VERSION}.${OpenCASCADE_MINOR_VERSION}.${OpenCASCADE_MAINTENANCE_VERSION}")
endif()
if(OpenCASCADE_FOUND)
    file(TO_CMAKE_PATH "${OpenCASCADE_RESOURCE_DIR}" _medforge_occt_resource)
    set(_medforge_occt_lib "${OpenCASCADE_LIBRARY_DIR}")
    set(_medforge_occt_bin "${OpenCASCADE_BINARY_DIR}")
    # A Debug-only Windows install puts import libraries in libd and DLLs in bind.
    if(EXISTS "${OpenCASCADE_INSTALL_PREFIX}/win64/vc14/libd/TKernel.lib")
        set(_medforge_occt_lib "${OpenCASCADE_INSTALL_PREFIX}/win64/vc14/libd")
        set(_medforge_occt_bin "${OpenCASCADE_INSTALL_PREFIX}/win64/vc14/bind")
    endif()
    target_include_directories(medforge_cad SYSTEM PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
    # The local kit is the Debug CRT build. Release stays on the stub.
    target_compile_definitions(medforge_cad PRIVATE "MEDFORGE_HAS_OCCT=$<CONFIG:Debug>")
    foreach(_medforge_occt_toolkit IN LISTS OpenCASCADE_LIBRARIES)
        if(WIN32)
            target_link_libraries(medforge_cad PUBLIC
                $<$<CONFIG:Debug>:${_medforge_occt_lib}/${_medforge_occt_toolkit}.lib>)
        else()
            target_link_libraries(medforge_cad PUBLIC
                $<$<CONFIG:Debug>:${_medforge_occt_lib}/lib${_medforge_occt_toolkit}.so>)
        endif()
    endforeach()
    set(MEDFORGE_OCCT_BIN "${_medforge_occt_bin}" PARENT_SCOPE)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/medforge_occt_resource.h"
"#pragma once\n#define MEDFORGE_OCCT_RESOURCE_DIR R\"(${_medforge_occt_resource})\"\n")
